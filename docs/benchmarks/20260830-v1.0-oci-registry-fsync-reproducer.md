## Executed Script

`0d9782af637de7770306cf6a0df8b866b8d9addbee040caa5a94c416be11e644`.

The positive and faulted sessions share the payload digest but have different
upload UUIDs. The second GET of a completed blob verifies only the positive
control. The faulted session is queried and cancelled, never completed.
Registry-private partial-file inspection below is diagnostic evidence only,
not a proposed SeqEvi integration interface.

```python
"""Disposable official-Registry HTTP durability probe, not a SeqEvi test gate.

Only subprocesses started here receive fault injection or termination signals.
Exit zero means the planned characterization ran, NOT Registry qualification.
"""

import hashlib
import json
import os
from pathlib import Path
import signal
import socket
import subprocess
import time
import urllib.error
import urllib.parse
import urllib.request


ROOT = Path(__file__).resolve().parent
PAYLOAD = b"hello, world!"
DIGEST = "sha256:" + hashlib.sha256(PAYLOAD).hexdigest()
REPOSITORY = "seqevi/unit0"
with socket.socket() as listener:
    listener.bind(("127.0.0.1", 0))
    PORT = listener.getsockname()[1]
BASE = f"http://127.0.0.1:{PORT}"
RESULT = {"payload_bytes": len(PAYLOAD), "payload_digest": DIGEST, "port": PORT}
EVENTS = []


def request(method, location, data=None, headers=None):
    url = urllib.parse.urljoin(BASE, location)
    if urllib.parse.urlsplit(url).netloc != urllib.parse.urlsplit(BASE).netloc:
        raise RuntimeError("probe Location escaped its loopback authority")
    req = urllib.request.Request(url, data=data, headers=headers or {}, method=method)
    try:
        response = urllib.request.urlopen(req, timeout=5)
    except urllib.error.HTTPError as exc:
        response = exc
    with response:
        body = response.read()
        result = {
            "status": response.status,
            "headers": dict(response.headers.items()),
            "body": body.decode("utf-8", errors="replace"),
        }
    if location != "/v2/":
        EVENTS.append({"method": method, "url": url, **result})
    return result


def header(response, name):
    return next(value for key, value in response["headers"].items() if key.lower() == name.lower())


def expect(response, status):
    if response["status"] != status:
        raise AssertionError(response)
    return response


def group_alive(pgid):
    try:
        os.killpg(pgid, 0)
    except ProcessLookupError:
        return False
    return True


class Registry:
    def __init__(self, name, inject=False):
        self.name = name
        self.log = (ROOT / f"{name}.log").open("wb")
        command = [str(ROOT / "registry"), "serve", str(ROOT / "config.yml")]
        if inject:
            command = [
                "strace", "-f", "-o", str(ROOT / "fsync.trace"),
                "-e", "trace=fsync", "-e", "inject=fsync:error=EIO",
                *command,
            ]
        environment = {
            **os.environ,
            "OTEL_TRACES_EXPORTER": "none",
            "REGISTRY_HTTP_ADDR": f"127.0.0.1:{PORT}",
            "GOMAXPROCS": "2",
        }
        self.process = subprocess.Popen(
            command, stdout=self.log, stderr=subprocess.STDOUT,
            env=environment, start_new_session=True,
        )

    def __enter__(self):
        try:
            deadline = time.monotonic() + 10
            while time.monotonic() < deadline:
                if self.process.poll() is not None:
                    raise RuntimeError(f"{self.name} exited; inspect {self.name}.log")
                try:
                    if request("GET", "/v2/")["status"] == 200:
                        return self
                except (urllib.error.URLError, ConnectionError, TimeoutError):
                    pass
                time.sleep(0.1)
            raise TimeoutError(f"{self.name} readiness")
        except BaseException:
            self.stop()
            raise

    def stop(self):
        pgid = self.process.pid
        if group_alive(pgid):
            os.killpg(pgid, signal.SIGTERM)
        try:
            self.process.wait(timeout=5)
        except subprocess.TimeoutExpired:
            os.killpg(pgid, signal.SIGKILL)
            self.process.wait(timeout=5)
        deadline = time.monotonic() + 5
        while group_alive(pgid) and time.monotonic() < deadline:
            time.sleep(0.05)
        if group_alive(pgid):
            raise RuntimeError(f"probe process group {pgid} not empty")
        self.log.close()
        RESULT.setdefault("released_process_groups", []).append(pgid)

    def __exit__(self, *args):
        self.stop()


def initiate():
    response = expect(request("POST", f"/v2/{REPOSITORY}/blobs/uploads/", b""), 202)
    return header(response, "Location"), header(response, "Docker-Upload-UUID")


def data_size(uuid):
    target = ROOT / "data/docker/registry/v2/repositories" / REPOSITORY / "_uploads" / uuid / "data"
    return target.stat().st_size


def patch(location):
    return request("PATCH", location, PAYLOAD, {
        "Content-Type": "application/octet-stream",
        "Content-Range": f"0-{len(PAYLOAD) - 1}",
    })


def completion_location(location):
    parts = urllib.parse.urlsplit(location)
    query = urllib.parse.parse_qsl(parts.query) + [("digest", DIGEST)]
    return urllib.parse.urlunsplit(parts._replace(query=urllib.parse.urlencode(query)))


def main():
    with Registry("positive"):
        normal_location, normal_uuid = initiate()
        normal_patch = expect(patch(normal_location), 202)
        RESULT["positive"] = {
            "patch_status": normal_patch["status"],
            "patch_range": header(normal_patch, "Range"),
            "partial_bytes_after_ack": data_size(normal_uuid),
        }
        assert RESULT["positive"]["partial_bytes_after_ack"] == len(PAYLOAD)
        expect(request("PUT", completion_location(header(normal_patch, "Location")), b""), 201)
        downloaded = expect(request("GET", f"/v2/{REPOSITORY}/blobs/{DIGEST}"), 200)
        assert downloaded["body"].encode() == PAYLOAD
        RESULT["positive"]["download_matches"] = True
        failing_location, failing_uuid = initiate()

    with Registry("injected", inject=True):
        bad_patch = patch(failing_location)
        RESULT["fault"] = {
            "patch_status": bad_patch["status"],
            "patch_range": header(bad_patch, "Range"),
            "partial_bytes_after_ack": data_size(failing_uuid),
        }
        recovery_location = header(bad_patch, "Location")

    trace = (ROOT / "fsync.trace").read_text()
    RESULT["fault"]["injected_eio_count"] = sum(
        "fsync(" in line and "EIO" in line and "INJECTED" in line
        for line in trace.splitlines()
    )
    assert RESULT["fault"]["injected_eio_count"] > 0

    with Registry("recovery"):
        status = expect(request("GET", recovery_location), 204)
        RESULT["recovery"] = {
            "status": status["status"],
            "range": header(status, "Range"),
            "partial_bytes": data_size(failing_uuid),
        }
        assert expect(request("GET", f"/v2/{REPOSITORY}/blobs/{DIGEST}"), 200)["body"].encode() == PAYLOAD
        expect(request("DELETE", header(status, "Location")), 204)
        RESULT["recovery"]["cancel_status"] = 204

    RESULT["durable_ack_counterexample"] = (
        RESULT["fault"]["patch_status"] == 202
        and RESULT["fault"]["patch_range"] == "0-12"
        and RESULT["fault"]["partial_bytes_after_ack"] == 0
        and RESULT["recovery"]["partial_bytes"] == 0
    )
    print(json.dumps(RESULT, indent=2))


try:
    main()
finally:
    (ROOT / "http-events.json").write_text(json.dumps(EVENTS, indent=2) + "\n")
    (ROOT / "result.json").write_text(json.dumps(RESULT, indent=2) + "\n")
```

## Expected Interpretation

The observed positive control was `202 / Range: 0-12`, partial size 13,
completion `201`, and exact downloaded-byte equality. The injected case was
`202 / Range: 0-12`, partial size 0, four injected `fsync` errors, and restart
status `204 / Range: 0-0` with partial size 0. Session cancellation was `204`.

If a successor returns an explicit error instead, this historical harness
may stop when the expected success headers are absent. Inspect the saved
response and trace; do not reinterpret a harness exception as proof that the
successor passes. Design its positive/negative acceptance check against the
same durable-ack requirement, including completion durability and cleanup.
